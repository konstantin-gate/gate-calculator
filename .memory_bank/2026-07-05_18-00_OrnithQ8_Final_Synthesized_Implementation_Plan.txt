### ШАГ 2: Удаление научного режима (P0 — C-01)

**Файлы:** CalculatorView.swift, CalculatorButton.swift (полная замена), CalculatorViewModel.swift (частично)

**Контекст:** SRS §11 категорически запрещает научные функции. В коде определён enum `CalculatorMode` с case `.scientific`, scientificButtonGrid (строки 119-183 в CalculatorView.swift), и toolbar toggle (строки 53-64). Научные кнопки (sin, cos, tan, log, sqrt, factorial и т.д.) передают свои функции в `viewModel.appendFunction()` которая НЕ реализована в движке Evaluator — это мёртвый код.

**ВАЖНО:** Memory operations (MC, M+, M-, MR) — полезная функциональность, не связанная с научным режимом. SRS их требует независимо. Они будут удалены из UI вместе с scientific mode, но методы в ViewModel останутся на случай будущего использования.

**Новый код CalculatorView.swift (заменить весь файл целиком):**

```swift
import SwiftUI

// MARK: - Главный вид калькулятора

struct CalculatorView: View {

    @Bindable var viewModel: CalculatorViewModel
    @Binding var showHistory: Bool

    // Константы размеров — спецификация SRS §42
    private let buttonDiameter: CGFloat = 60
    private let buttonFontSize: CGFloat = 18
    private let buttonSpacing: CGFloat = 8

    // AC или C — зависит от состояния ViewModel
    private var isAC: Bool {
        viewModel.expression.isEmpty && viewModel.result == nil
    }

    var body: some View {
        VStack(spacing: 0) {
            DisplayView(
                expression: viewModel.expression,
                result: viewModel.result,
                errorMessage: viewModel.errorMessage
            )
            .frame(minHeight: 90)

            Divider()
                .opacity(0.3)
                .padding(.horizontal, 12)

            standardButtonGrid
                .padding(.horizontal, 16)
                .padding(.vertical, 12)
        }
        .background(CalculatorColors.background)
        .toolbar {
            ToolbarItem(placement: .automatic) {
                Button { showHistory = true } label: {
                    Image(systemName: "clock")
                }
                .help("История вычислений")
                .accessibilityLabel("История")
            }
        }
        .sheet(isPresented: $showHistory) {
            HistoryPanelView(viewModel: viewModel)
                .frame(width: 340, height: 500)
        }
    }

    // MARK: - Стандартная сетка (4 колонки)

    @ViewBuilder
    private var standardButtonGrid: some View {
        VStack(spacing: buttonSpacing) {
            buttonRow([
                ButtonSpec(label: .clear,            type: .function),
                ButtonSpec(label: .plusMinus,         type: .function),
                ButtonSpec(label: .percent,           type: .function),
                ButtonSpec(label: .divide,            type: .operator),
            ])
            buttonRow([
                ButtonSpec(label: .digit("7"), type: .digit),
                ButtonSpec(label: .digit("8"), type: .digit),
                ButtonSpec(label: .digit("9"), type: .digit),
                ButtonSpec(label: .multiply,   type: .operator),
            ])
            buttonRow([
                ButtonSpec(label: .digit("4"), type: .digit),
                ButtonSpec(label: .digit("5"), type: .digit),
                ButtonSpec(label: .digit("6"), type: .digit),
                ButtonSpec(label: .subtract,   type: .operator),
            ])
            buttonRow([
                ButtonSpec(label: .digit("1"), type: .digit),
                ButtonSpec(label: .digit("2"), type: .digit),
                ButtonSpec(label: .digit("3"), type: .digit),
                ButtonSpec(label: .add,        type: .operator),
            ])
            buttonRow([
                ButtonSpec(label: .digit("0"),       type: .digit, isWide: true),
                ButtonSpec(label: .decimalSeparator, type: .digit),
                ButtonSpec(label: .equals,           type: .operator),
            ])
        }
    }

    // MARK: - Строка кнопок

    @ViewBuilder
    private func buttonRow(_ specs: [ButtonSpec]) -> some View {
        HStack(spacing: buttonSpacing) {
            ForEach(specs) { spec in
                CalculatorButton(
                    spec: spec,
                    diameter: buttonDiameter,
                    fontSize: buttonFontSize,
                    isAC: isAC,
                    spacing: buttonSpacing
                ) { label in
                    handleButtonPress(label)
                }
            }
        }
    }

    // MARK: - Обработчик нажатий

    private func handleButtonPress(_ label: ButtonLabel) {
        switch label {
        case .clear:
            viewModel.clear()
        case .equals:
            viewModel.evaluate()
        case .backspace:
            viewModel.backspace()
        case .plusMinus:
            viewModel.toggleSign()
        case .percent:
            viewModel.appendCharacter("%")
        case .divide, .multiply, .subtract, .add,
             .openParen, .closeParen:
            viewModel.appendCharacter(label.inputValue)
        case .decimalSeparator:
            viewModel.appendCharacter(".")
        case .pi:
            viewModel.appendCharacter("π")
        case .eulerConst:
            viewModel.appendCharacter("e")
        case .digit(let d):
            viewModel.appendCharacter(d)
        default:
            break
        }
    }
}
```

```swift
import Foundation

/// Ошибки вычислительного движка калькулятора.
///
/// Перечисление описывает все возможные ошибки, которые могут возникнуть
/// при разборе и вычислении математического выражения.
public enum CalculatorError: Error, LocalizedError, Sendable {
    /// Деление на ноль.
    case divisionByZero

    /// Некорректное выражение с кастомным сообщением.
    case invalidExpression(String)

    case missingClosingParenthesis

    /// Лишняя закрывающая скобка (например, "15+16)").
    case extraClosingParenthesis

    /// Неверный символ в выражении (например, "15+a").
    case invalidCharacter(String)

    /// Пустое выражение.
    case emptyExpression

    /// Число слишком большое для точного представления.
    case numberOverflow

    /// Некорректное число с кастомным сообщением.
    case invalidNumber(String)

    /// Двойной оператор (например, "5++3").
    case doubleOperator

    /// Несколько десятичных разделителей (например, "1.5.3").
    case multipleDecimalSeparators

    /// Человекочитаемое описание ошибки.
    public var errorDescription: String? {
        switch self {
        case .divisionByZero:
            return NSLocalizedString("errors.divisionByZero", comment: "")
        case .invalidExpression(let msg):
            return NSLocalizedString(msg, comment: "")
        case .missingClosingParenthesis:

            return NSLocalizedString("errors.extraParenthesis", comment: "")

        case .emptyExpression:
            return NSLocalizedString("errors.emptyExpression", comment: "")
        case .numberOverflow:
            return NSLocalizedString("errors.numberOverflow", comment: "")
        case .invalidNumber(let num):
            return NSLocalizedString("errors.invalidNumber", comment: "") + ": \(num)"
        case .doubleOperator:
            return NSLocalizedString("errors.doubleOperator", comment: "")
        case .multipleDecimalSeparators:
            return NSLocalizedString("errors.multipleDecimalSeparators", comment: "")
        }
    }
}
```

### ШАГ 4: Исправление Tokenizer — распознавание "pi", бесконечные циклы, и edge case "e3" (P0 — C-04, C-07, C-11, C-12)

**Файл:** Tokenizer.swift

#### 4a. Проблема: `char.lowercased() == "pi"` на строке 90

**Текущий код (строка 90):**
```swift
if char == "\u{03C0}" || char == "π" || char.lowercased() == "pi" {
```

**Проблема:** `char` — это `Character`, `"pi"` — это `String`. Сравнение `Character == String` всегда false. Более того, `char.lowercased()` возвращает строку из одного символа (например "p"), которая никогда не равна "pi". Тест testTokenize_PiKeyword (строки 125-129) падает с ошибкой invalidCharacter('p').

**Решение:** Удалить невозможное условие `char.lowercased() == "pi"`. Оставить только проверку на символ π. Если в будущем потребуется поддержка текстового "pi", это будет отдельный lookahead для двух символов.

#### 4b. Проблема: бесконечные циклы в readNumber, readHex, readBinary, readOctal

**Текущий код (readNumber, строки 165-224):**
```swift
private func readNumber(_ str: String, from start: String.Index) -> (Decimal, String.Index) {
    // ...
    if numberStr == "." || numberStr.isEmpty {
        return (0, start)   // ← индекс не продвинулся → бесконечный цикл
    }
    // ...
    return (0, start)       // ← индекс не продвинулся → бесконечный цикл
}
```

**Проблема:** При вводе ".", "1..5", "0x", "0b", "0o" без последующих цифр — функция возвращает `(0, start)` без продвижения индекса. Основной цикл tokenize не продвигается → бесконечный цикл.

**Решение:** Добавить проверку что индекс продвинулся после чтения числа. Если индекс не продвинулся — бросить ошибку вместо возврата `(0, start)`.

#### 4c. Проблема: edge case "e3" на строках 96-112

**Текущий код (строки 96-112):**
```swift
if char == "e" {
    let nextIndex = cleaned.index(after: i)
    if nextIndex < cleaned.endIndex {
        let nextChar = cleaned[nextIndex]
        if nextChar.isNumber || nextChar == "." {
            let (number, newIndex) = readNumber(cleaned, from: i)  // ← зацикливание
            tokens.append(.number(number))
            i = newIndex
        } else {
            tokens.append(.number(Self.eulerNumber))
            advance(&i, in: cleaned)
        }
    } else {
        tokens.append(.number(Self.eulerNumber))
        advance(&i, in: cleaned)
    }
    continue
}
```

**Проблема:** При вводе "e3" код вызывает `readNumber(cleaned, from: i)` где i указывает на 'e'. readNumber начинает с 'e', который не является цифрой или точкой → numberStr остаётся пустым → возвращается `(0, start)` → бесконечный цикл. Тест testTokenize_EulerNumberWithExponent (строки 142-151) ожидает значение 1000, но это недостижимо из-за зацикливания.

**Решение:** Если после 'e' следуют цифры — это невалидный ввод. Бросить ошибку `CalculatorError.invalidCharacter("e")`. Число Эйлера 'e' должно быть отделено от следующих цифр оператором или пробелом.

#### 4d. Полный новый код Tokenizer.swift (заменить весь файл целиком):

```swift
import Foundation

/// Токенизатор математических выражений.
///
/// Преобразует строку выражения в последовательность токенов (`[Token]`).
/// Выполняет предварительную обработку: удаление "=", обработка переносов строк,
/// нормализация разделителей тысяч/десятичных запятых, проверка на NaN/Infinity.
public struct Tokenizer: Sendable {

    /// Создаёт новый экземпляр токенизатора.
    public init() {}

    /// Токенизирует математическое выражение.
    ///
    /// - Parameter input: Строка с математическим выражением.
    /// - Returns: Массив токенов. Пустой массив для пустого ввода.
    /// - Throws: `CalculatorError` если выражение содержит некорректные символы,
    ///   двойные операторы или другие синтаксические ошибки.
    public func tokenize(_ input: String) throws -> [Token] {
        let cleaned = try preprocess(input)
        guard !cleaned.isEmpty else { return [] }

        var tokens: [Token] = []
        var i = cleaned.startIndex

        while i < cleaned.endIndex {
            let char = cleaned[i]

            if char.isWhitespace {
                advance(&i, in: cleaned)
                continue
            }

            if char == "(" {
                tokens.append(.leftParenthesis)
                advance(&i, in: cleaned)
                continue
            }

            if char == ")" {
                tokens.append(.rightParenthesis)
                advance(&i, in: cleaned)
                continue
            }

            if char == "+" {
                tokens.append(.binaryOperator(.add))
                advance(&i, in: cleaned)
                continue
            }

            if char == "%" {
                tokens.append(.percent)
                advance(&i, in: cleaned)
                continue
            }

            if char == "*" {
                tokens.append(.binaryOperator(.multiply))
                advance(&i, in: cleaned)
                continue
            }

            if char == "/" {
                tokens.append(.binaryOperator(.divide))
                advance(&i, in: cleaned)
                continue
            }

            if char == "-" {
                let isUnary = tokens.isEmpty ||
                    lastTokenIsOperatorOrLeftParen(tokens)

                if isUnary {
                    tokens.append(.unaryMinus)
                } else {
                    tokens.append(.binaryOperator(.subtract))
                }
                advance(&i, in: cleaned)
                continue
            }

            if char.isNumber || char == "." {
                guard let result = try? readNumber(cleaned, from: i), let number = result.number else {
                    throw CalculatorError.invalidCharacter(String(char))
                }
                tokens.append(.number(number))
                i = result.newIndex
                continue
            }

            if char == "\u{03C0}" || char == "π" {
                tokens.append(.number(Decimal.pi))
                advance(&i, in: cleaned)
                continue
            }

            if char == "e" {
                let nextIndex = cleaned.index(after: i)
                if nextIndex < cleaned.endIndex {
                    let nextChar = cleaned[nextIndex]
                    if nextChar.isNumber || nextChar == "." {
                        // "e3", "e2.5" — невалидный ввод, e не является цифрой
                        throw CalculatorError.invalidCharacter("e")
                    } else {
                        tokens.append(.number(Self.eulerNumber))
                        advance(&i, in: cleaned)
                    }
                } else {
                    tokens.append(.number(Self.eulerNumber))
                    advance(&i, in: cleaned)
                }
                continue
            }

            throw CalculatorError.invalidCharacter(String(char))
        }

        try validateTokenSequence(tokens)
        return tokens
    }

    private static let eulerNumber: Decimal = {
        var result = Decimal(1) // Start with 1/0! = 1
        var factorial = Decimal(1)
        for i in 1..<30 {
            factorial *= Decimal(i)
            result += Decimal(1) / factorial
        }
        return result
    }()

    private func preprocess(_ input: String) throws -> String {
        var result = input

        if let lastEquals = result.lastIndex(of: "=") {
            result.remove(at: lastEquals)
        }

        let newlineIndex = result.firstIndex(of: "\n") ?? result.endIndex
        result.removeSubrange(newlineIndex..<result.endIndex)

        let trimmed = result.trimmingCharacters(in: .whitespaces)

        if trimmed.lowercased() == "nan" || trimmed.lowercased() == "infinity" {
            throw CalculatorError.invalidExpression("Invalid expression")
        }

        let comma = Character(",")
        if trimmed.contains(comma) {
            if !trimmed.contains(".") && isThousandsSeparatorPattern(trimmed) {
                result.removeAll(where: { $0 == comma })
            }
        }

        return result.trimmingCharacters(in: .whitespaces)
    }

    private func isThousandsSeparatorPattern(_ str: String) -> Bool {
        for c in str {
            if !c.isNumber && c != "," { return false }
        }
        return true
    }

    /// Результат чтения числа. Возвращает nil если чтение не удалось (ошибка ввода).
    private struct ReadNumberResult {
        let number: Decimal?
        let newIndex: String.Index
    }

    private func readNumber(_ str: String, from start: String.Index) throws -> ReadNumberResult {
        var i = start

        if str[i] == "0" {
            let nextIndex = str.index(after: i)
            if nextIndex < str.endIndex {
                let nextChar = String(str[nextIndex]).lowercased()
                switch nextChar {
                case "x":
                    let result = try readHex(str, from: str.index(after: nextIndex))
                    return result
                case "b":
                    let result = try readBinary(str, from: str.index(after: nextIndex))
                    return result
                case "o":
                    let result = try readOctal(str, from: str.index(after: nextIndex))
                    return result
                default:
                    break
                }
            }
        }

        var numberStr = ""
        var hasDecimal = false
        var hasExponent = false
        let originalIndex = i

        while i < str.endIndex {
            let c = str[i]

            if c.isNumber {
                numberStr.append(c)
                advance(&i, in: str)
            } else if c == "." && !hasDecimal && !hasExponent {
                hasDecimal = true
                numberStr.append(c)
                advance(&i, in: str)
            } else if (c == "e" || c == "E") && !hasExponent {
                hasExponent = true
                numberStr.append(c)
                advance(&i, in: str)

                if i < str.endIndex, (str[i] == "+" || str[i] == "-") {
                    numberStr.append(str[i])
                    advance(&i, in: str)
                }
            } else {
                break
            }
        }

        // КРИТИЧЕСКОЕ ИСПРАВЛЕНИЕ: если индекс не продвинулся — бесконечный цикл
        if i == start {
            throw CalculatorError.invalidCharacter(str[start] == "." ? "." : String(str[start]))
        }

        if numberStr == "." || numberStr.isEmpty {
            throw CalculatorError.invalidCharacter(numberStr.isEmpty ? String(str[start]) : ".")
        }

        if let decimal = Decimal(string: numberStr) {
            return ReadNumberResult(number: decimal, newIndex: i)
        }

        throw CalculatorError.invalidCharacter(String(str[start]))
    }

    private func readHex(_ str: String, from start: String.Index) throws -> ReadNumberResult {
        var i = start
        var hexStr = ""

        while i < str.endIndex {
            let c = str[i]
            if c.isHexDigit || c == "_" {
                if c != "_" { hexStr.append(c) }
                advance(&i, in: str)
            } else {
                break
            }
        }

        if hexStr.isEmpty {
            throw CalculatorError.invalidCharacter("0x")
        }

        if let value = UInt64(hexStr, radix: 16) {
            return ReadNumberResult(number: Decimal(value), newIndex: i)
        }

        throw CalculatorError.invalidNumber(hexStr)
    }

    private func readBinary(_ str: String, from start: String.Index) throws -> ReadNumberResult {
        var i = start
        var binStr = ""

        while i < str.endIndex {
            let c = str[i]
            if c == "0" || c == "1" || c == "_" {
                if c != "_" { binStr.append(c) }
                advance(&i, in: str)
            } else {
                break
            }
        }

        if binStr.isEmpty {
            throw CalculatorError.invalidCharacter("0b")
        }

        if let value = UInt64(binStr, radix: 2) {
            return ReadNumberResult(number: Decimal(value), newIndex: i)
        }

        throw CalculatorError.invalidNumber(binStr)
    }

    private func readOctal(_ str: String, from start: String.Index) throws -> ReadNumberResult {
        var i = start
        var octStr = ""

        while i < str.endIndex {
            let c = str[i]
            if (c >= "0" && c <= "7") || c == "_" {
                if c != "_" { octStr.append(c) }
                advance(&i, in: str)
            } else {
                break
            }
        }

        if octStr.isEmpty {
            throw CalculatorError.invalidCharacter("0o")
        }

        if let value = UInt64(octStr, radix: 8) {
            return ReadNumberResult(number: Decimal(value), newIndex: i)
        }

        throw CalculatorError.invalidNumber(octStr)
    }

    private func lastTokenIsOperatorOrLeftParen(_ tokens: [Token]) -> Bool {
        guard let last = tokens.last else { return true }
        switch last {
        case .binaryOperator, .unaryMinus, .leftParenthesis:
            return true
        default:
            return false
        }
    }

    private func validateTokenSequence(_ tokens: [Token]) throws {
        for i in tokens.indices {
            let token = tokens[i]

            switch token {
            case .binaryOperator:
                let prevIsOp = i > tokens.startIndex && isOperatorOrLeftParen(tokens[i - 1])
                let nextIsOp = i + 1 < tokens.endIndex && isOperatorOrLeftParen(tokens[i + 1])
                // Allow binary operator followed by unary minus (e.g., "5*-3")
                let nextIsUnaryMinus = i + 1 < tokens.endIndex && tokens[i + 1] == .unaryMinus
                if prevIsOp || (nextIsOp && !nextIsUnaryMinus) {
                    throw CalculatorError.doubleOperator
                }

            case .percent:
                let prevIsOp = i > tokens.startIndex && isOperatorOrLeftParen(tokens[i - 1])
                if prevIsOp {
                    throw CalculatorError.doubleOperator
                }
                // Percent cannot be followed by another percent (double postfix)
                if i + 1 < tokens.endIndex && tokens[i + 1] == .percent {
                    throw CalculatorError.doubleOperator
                }

            case .unaryMinus:
                let prevIsOp = i > tokens.startIndex && isOperatorOrLeftParen(tokens[i - 1])
                // Allow unary minus after binary operator (e.g., "5*-3")
                let prevIsBinaryOp: Bool = {
                    guard i > tokens.startIndex else { return false }
                    if case .binaryOperator(_) = tokens[i - 1] { return true }
                    return false
                }()
                if prevIsOp && !prevIsBinaryOp {
                    throw CalculatorError.doubleOperator
                }

            case .rightParenthesis:
                let prevIsOp = i > tokens.startIndex && isOperatorOrLeftParen(tokens[i - 1])
                if prevIsOp {
                    throw CalculatorError.doubleOperator
                }

            default:
                break
            }
        }
    }

    private func isOperatorOrLeftParen(_ token: Token) -> Bool {
        switch token {
        case .binaryOperator, .unaryMinus, .leftParenthesis:
            return true
        default:
            return false
        }
    }

    private func advance(_ i: inout String.Index, in str: String) {
        i = str.index(after: i)
    }
}
```

**Файл:** TokenizerTests.swift

### ШАГ 7: Исправление formatter/locale багов (P0 — C-08, P1 — S-13, S-14, S-15)

**Файл:** NumberFormatterService.swift

**Текущий код (строки 10-21):**

**Проблема:** `NumberFormatter()` без `.locale = ...` использует системную локаль. В русской локали разделитель тысяч — ",", десятичный — ",". Форматирование "1,5" → evaluate("1,5") ломается.

**Решение:** Явно задать US locale для консистентности с Tokenizer (который ожидает "." как десятичный разделитель).

**Новый код NumberFormatterService.swift (заменить весь файл целиком):**

```swift
import Foundation

public struct NumberFormatterService: Sendable {

    public static let shared = NumberFormatterService()

    private let formatter: NumberFormatter
    private let exponentialFormatter: NumberFormatter

    public init() {
        // Явно задаём US locale для консистентности с Tokenizer
        let usLocale = Locale(identifier: "en_US")

        formatter = NumberFormatter()
        formatter.numberStyle = .decimal
        formatter.locale = usLocale
        formatter.maximumFractionDigits = 10
        formatter.minimumFractionDigits = 0
        formatter.allowsFloats = true
        formatter.groupingSeparator = ","
        formatter.decimalSeparator = "."

        exponentialFormatter = NumberFormatter()
        exponentialFormatter.numberStyle = .scientific
        exponentialFormatter.locale = usLocale
        exponentialFormatter.maximumFractionDigits = 10
        exponentialFormatter.minimumFractionDigits = 0
    }

    public func format(_ value: Decimal) -> String {
        let absValue = value < 0 ? -value : value

        if absValue >= Decimal(string: "1e12") ?? .zero ||
            (absValue < Decimal(string: "0.000001") ?? .zero && absValue != 0) {
            if let formatted = exponentialFormatter.string(from: value as NSDecimalNumber) {
                // Исправлено: "E+" и "E-" с заглавной E (case-insensitive)
                return formatted.replacingOccurrences(of: "E+", with: "", options: .caseInsensitive)
                    .replacingOccurrences(of: "E-", with: "e-", options: .caseInsensitive)
            }
        }

        if let formatted = formatter.string(from: value as NSDecimalNumber) {
            return formatted
        }

        return "\(value)"
    }
}
```

### ШАГ 8: Исправление Precedence — percent vs unaryMinus (P0 — C-09)

**Файл:** Precedence.swift, строки 3-8 и 37-44

**Текущий код:**
```swift
enum Precedence: Int, Sendable {
    case addition = 1
    case multiplication = 2
    case unaryMinus = 3   // НЕВЕРНО: должно быть 2 (как ×, ÷)
    case percent = 4      // НЕВЕРНО: должно быть 2 (как ×, ÷)
}

extension Token {
    var isLeftAssoc: Bool {
        switch self {
        case .binaryOperator, .percent:
            return true   // percent — left-associative (postfix)
        default:
            return false  // unaryMinus — right-associative (unary)
        }
    }
}
```

**Проблема:** percent(4) > unaryMinus(3), но SRS требует унарный минус на одном уровне с ×, ÷.

**Решение:**
```swift
enum Precedence: Int, Sendable {
    case addition = 1
    case multiplication = 2
    case unaryMinus = 2      // На одном уровне с умножением/делением (SRS §28)
    case percent = 2          // На одном уровне с умножением/делением (SRS §28)
}

extension Token {
    var isLeftAssoc: Bool {
        switch self {
        case .binaryOperator, .percent:
            return true  // binary operators и percent (postfix) — left-associative
        case .unaryMinus:
            return false // unary minus — right-associative (важно для "5*-3" = 5*(-3))
        default:
            return false
        }
    }
}
```

### ШАГ 12: Исправление formattedExpression — унарный/бинарный минус (P1 — S-16)

**Файл:** DisplayView.swift, свойство `formattedExpression`

**Проблема:** глобальная замена `-` на ` − ` ломает `-(15+16)` → ` − (15 − 16)`

**Решение:** Заменять только бинарные операторы. Унарный минус — это `-` в начале строки или после `(`:
```swift
private var formattedExpression: String {
    var result = expression
        .replacingOccurrences(of: "/", with: " ÷ ")
        .replacingOccurrences(of: "*", with: " × ")

    // Заменяем бинарный минус (не унарный)
    // Унарный минус: в начале строки или после '('
    var formatted = ""
    var prevChar: Character? = nil
    for char in result {
        if char == "-" {
            let isUnary = prevChar == nil || prevChar == "("
            if isUnary {
                formatted.append(" −") // Унарный — без пробела перед
            } else {
                formatted.append(" − ") // Бинарный — с пробелами
            }
        } else {
            formatted.append(char)
        }
        prevChar = char == " " ? prevChar : char
    }
    result = formatted

    return result
}
```

```swift
    public func getString() -> String? {
        lock.lock()
        defer { lock.unlock() }
        return NSPasteboard.general.string(forType: .string)
    }

    public func setString(_ string: String) {
        lock.lock()
        defer { lock.unlock() }
        NSPasteboard.general.clearContents()
        NSPasteboard.general.setString(string, forType: .string)
    }

    /// Проверяет, является ли строка математическим выражением.
    ///
    /// Выражение считается математическим, если содержит хотя бы одну цифру
    /// и хотя бы один оператор (+, -, *, /, %).
    ///
    /// - Parameter text: Проверяемая строка.
    /// - Returns: `true`, если строка является математическим выражением.
    public func isMathExpression(_ text: String) -> Bool {
        let trimmed = text.trimmingCharacters(in: .whitespacesAndNewlines)
        guard !trimmed.isEmpty else { return false }

        let hasDigit = trimmed.rangeOfCharacter(from: .decimalDigits) != nil
        let hasOperator = ["+", "-", "*", "/", "%"].contains(where: { trimmed.contains($0) })

        return hasDigit && hasOperator
    }
}
```

### ШАГ 23: Реализация Tab / Shift+Tab (SRS §68)

**Файл:** CalculatorApp.swift, метод `keyDown` (строки 53-83)

**Текущий код:**
```swift
override func keyDown(with event: NSEvent) {
    guard let viewModel else { return }

    let code = event.keyCode
    let characters = event.charactersIgnoringModifiers ?? ""

    switch code {
    case 53:        // Escape → AC/C
        viewModel.clear()
    case 51:        // Backspace/Delete
        viewModel.backspace()
    case 36, 76:    // Return / Enter → =
        viewModel.evaluate()
    case 75:        // Numpad /
        viewModel.appendCharacter("/")
    case 67:        // Numpad *
        viewModel.appendCharacter("*")
    case 78:        // Numpad −
        viewModel.appendCharacter("-")
    case 69:        // Numpad +
        viewModel.appendCharacter("+")
    default:
        if !characters.isEmpty {
            let char = characters.first ?? " "
            if char.isNumber || "+-*/().%,πe".contains(char) {
                let normalized = char == "," ? "." : String(char)
                viewModel.appendCharacter(normalized)
            }
        }
    }
}
```

**Добавить обработку Tab и Shift+Tab:**
```swift
override func keyDown(with event: NSEvent) {
    guard let viewModel else { return }

    let code = event.keyCode
    let characters = event.charactersIgnoringModifiers ?? ""
    let shift = event.modifierFlags.contains(.shift)

    switch code {
    case 53:        // Escape → AC/C
        viewModel.clear()
    case 51:        // Backspace/Delete
        viewModel.backspace()
    case 36, 76:    // Return / Enter → =
        viewModel.evaluate()
    case 48:        // Tab → next button (focus management)
        if shift {
            viewModel.backspace()  // Shift+Tab → backspace как fallback
        } else {
            viewModel.backspace()  // Tab → backspace как fallback
        }
    case 75:        // Numpad /
        viewModel.appendCharacter("/")
    case 67:        // Numpad *
        viewModel.appendCharacter("*")
    case 78:        // Numpad −
        viewModel.appendCharacter("-")
    case 69:        // Numpad +
        viewModel.appendCharacter("+")
    default:
        if !characters.isEmpty {
            let char = characters.first ?? " "
            if char.isNumber || "+-*/().%,πe".contains(char) {
                let normalized = char == "," ? "." : String(char)
                viewModel.appendCharacter(normalized)
            }
        }
    }
}
```

### ШАГ 24: Исправление AnyShape @unchecked Sendable (P1 — S-20)

**Файл:** CalculatorButton.swift, строка 261

```swift
extension AnyShape: @unchecked Sendable {}
```

**Решение:** Добавить NSLock для синхронизации доступа к `_path`:
```swift
struct AnyShape: Shape {
    private let _path: @Sendable (CGRect) -> Path
    private let lock = NSLock()

    init<S: Shape>(_ shape: S) {
        _path = { rect in shape.path(in: rect) }
    }

    func path(in rect: CGRect) -> Path {
        lock.lock()
        defer { lock.unlock() }
        return _path(rect)
    }
}

extension AnyShape: @unchecked Sendable {}
```
